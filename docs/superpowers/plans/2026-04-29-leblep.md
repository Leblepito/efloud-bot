```python
"""Cross-platform path resolution for Leblep state and code directories."""
from __future__ import annotations

from pathlib import Path

import leblep.paths as _self  # for monkeypatch-friendly module-level access

LEBLEP_STATE_DIR: Path = Path.home() / ".leblep"
LEBLEP_CODE_DIR: Path = Path(__file__).resolve().parent.parent

STATE_SUBDIRS: tuple[tuple[str, ...], ...] = (
    ("memory", "user"),
    ("memory", "feedback"),
    ("memory", "project"),
    ("memory", "reference"),
    ("skills", "pinned"),
    ("skills", "absorbed"),
    ("skills", "trial"),
    ("skills", ".trash"),
    ("patterns",),
    ("commands",),
    ("logs",),
)


def expand(p: str | Path) -> Path:
    return Path(p).expanduser().resolve()


def state_path(*parts: str) -> Path:
    return _self.LEBLEP_STATE_DIR.joinpath(*parts)


def ensure_state_dirs() -> None:
    base = _self.LEBLEP_STATE_DIR
    for parts in STATE_SUBDIRS:
        base.joinpath(*parts).mkdir(parents=True, exist_ok=True)
    gitignore = base / ".gitignore"
    if not gitignore.exists():
        gitignore.write_text("*\n")
```

> **`import leblep.paths as _self`** is the standard idiom for letting `monkeypatch.setattr("leblep.paths.LEBLEP_STATE_DIR", ...)` flow into module-level functions. Without it, function-local `LEBLEP_STATE_DIR` resolves at import time and won't see the patch.

`scripts/extract_legacy_dict.py`:
```python
"""Extract a top-level dict literal (PATTERNS or SKILL_REGISTRY) from legacy
superagentv3.py via AST and either emit it to stdout or rewrite it in-place
between BEGIN_AUTOGEN / END_AUTOGEN markers in a target file.

Handles BOTH ast.Assign (`PATTERNS = {...}`) and ast.AnnAssign
(`SKILL_REGISTRY: Dict[str, Dict] = {...}`).

Usage:
    # Emit to stdout (manual paste workflow)
    python scripts/extract_legacy_dict.py PATTERNS

    # Rewrite leblep/seed/patterns.py in-place between markers
    python scripts/extract_legacy_dict.py PATTERNS --inplace leblep/seed/patterns.py

    # Same for SKILL_REGISTRY
    python scripts/extract_legacy_dict.py SKILL_REGISTRY --inplace leblep/seed/skills.py
"""
from __future__ import annotations

import argparse
import ast
import pprint
import re
import sys
from pathlib import Path

LEGACY = Path("c:/Users/utkuc/Downloads/efloud-bot/superagentv3.py")

_BEGIN = "# BEGIN_AUTOGEN"
_END = "# END_AUTOGEN"


def find_dict_literal(tree: ast.AST, name: str) -> dict | None:
    """Find a top-level dict literal assigned to `name`.

    Supports both:
      - ast.Assign:    `name = {...}`              (e.g. PATTERNS)
      - ast.AnnAssign: `name: Type = {...}`        (e.g. SKILL_REGISTRY)
    """
    for node in tree.body:
        if isinstance(node, ast.Assign) and len(node.targets) == 1:
            t = node.targets[0]
            if isinstance(t, ast.Name) and t.id == name:
                return ast.literal_eval(node.value)
        elif isinstance(node, ast.AnnAssign) and isinstance(node.target, ast.Name):
            if node.target.id == name and node.value is not None:
                return ast.literal_eval(node.value)
    return None


def render_block(name: str, d: dict, type_hint: str) -> str:
    """Format the dict as a Python source block ready for pasting/replacing."""
    body = pprint.pformat(d, width=120, sort_dicts=False)
    return f"{name}: {type_hint} = {body}\n"


def rewrite_inplace(target: Path, name: str, block: str) -> None:
    """Replace content between BEGIN_AUTOGEN / END_AUTOGEN markers in target file."""
    if not target.exists():
        raise FileNotFoundError(target)
    text = target.read_text(encoding="utf-8")

    pattern = re.compile(
        rf"({re.escape(_BEGIN)}.*?\n)(.*?)({re.escape(_END)})",
        re.DOTALL,
    )
    if not pattern.search(text):
        raise ValueError(
            f"{target} missing BEGIN_AUTOGEN/END_AUTOGEN markers — cannot --inplace"
        )

    def replace(m: re.Match) -> str:
        return f"{m.group(1)}{block}{m.group(3)}"

    new_text = pattern.sub(replace, text, count=1)
    target.write_text(new_text, encoding="utf-8")


def main() -> int:
    parser = argparse.ArgumentParser(description="Extract legacy dict via AST")
    parser.add_argument("name", choices=["PATTERNS", "SKILL_REGISTRY"])
    parser.add_argument("--inplace", type=Path, help="Rewrite target file between markers")
    parser.add_argument(
        "--type-hint",
        default=None,
        help="Override emitted type annotation (default depends on name)",
    )
    args = parser.parse_args()

    if not LEGACY.exists():
        print(f"ERROR: legacy file not found at {LEGACY}", file=sys.stderr)
        return 1

    tree = ast.parse(LEGACY.read_text(encoding="utf-8"))
    d = find_dict_literal(tree, args.name)
    if d is None:
        print(f"ERROR: {args.name} not found in {LEGACY}", file=sys.stderr)
        return 1

    type_hint = args.type_hint or {
        "PATTERNS": "dict[str, dict[str, str]]",
        "SKILL_REGISTRY": "dict[str, dict[str, object]]",
    }[args.name]

    block = render_block(args.name, d, type_hint)

    print(f"# Extracted {len(d)} entries for {args.name} from {LEGACY}", file=sys.stderr)

    if args.inplace:
        rewrite_inplace(args.inplace, args.name, block)
        print(f"Rewrote {args.inplace} in-place ({len(d)} entries)", file=sys.stderr)
    else:
        sys.stdout.write(block)

    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""Migrate the legacy PATTERNS dict to ~/.leblep/patterns/<source-slug>/<name>.md.

PATTERNS auto-populated by Task 5's extractor:
    python scripts/extract_legacy_dict.py PATTERNS --inplace leblep/seed/patterns.py
Re-run after the legacy file changes.
"""
from __future__ import annotations

import re
from datetime import datetime
from pathlib import Path

import yaml

import leblep.seed.patterns as _self
from leblep.paths import state_path

_OUT_BASE: Path = state_path("patterns")


# BEGIN_AUTOGEN
PATTERNS: dict[str, dict[str, str]] = {
    # Auto-populated by: python scripts/extract_legacy_dict.py PATTERNS --inplace leblep/seed/patterns.py
}
# END_AUTOGEN


def _slugify(s: str) -> str:
    return re.sub(r"[^\w\-]", "-", s.lower()).strip("-")


def run_seed() -> int:
    """Write each pattern to its source-slugged dir. Returns count of files written.

    Idempotent: timestamps in frontmatter update each run, but bodies stay byte-equal.
    """
    base = _self._OUT_BASE
    written = 0
    for name, p in PATTERNS.items():
        source_slug = _slugify(p["source"])
        out_dir = base / source_slug
        out_dir.mkdir(parents=True, exist_ok=True)
        out = out_dir / f"{name}.md"

        frontmatter = {
            "name": name,
            "source": p["source"],
            "description": p["description"],
            "imported_at": datetime.now().isoformat(),
        }
        body = p["pattern"]
        content = (
```

```python
def _gh_fetch_file(repo: str, path: str) -> str:
    result = subprocess.run(
        ["gh", "api", f"repos/{repo}/contents/{path}", "--jq", ".content"],
        capture_output=True,
        text=True,
        check=True,
    )
    encoded = result.stdout.strip()
    return base64.b64decode(encoded).decode("utf-8")

            logger.warning("Skipping %s: %s", target["path"], e)
```

```bash
from pathlib import Path
```

```python
def _resolve_pattern_path(pattern_name: str) -> str:
    """Best-effort: return canonical state-dir-relative path for a pattern.
```

## Implementation Strategy

- **Chunk 1** is fully detailed (lines 17-1454), ready to execute now via `superpowers:subagent-driven-development` (paralel worktree subagent).
- **Chunks 2-6** outline the tasks; **detailed TDD steps for each chunk are drafted just-in-time at the start of that chunk's execution**, applying the same review loop pattern (write chunk → reviewer → fix → approve → execute).
- This avoids drafting 6000+ lines of plan upfront when later chunks may need adjustments based on Chunk N-1 execution learnings.
